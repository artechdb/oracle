SELECT trunc(sysdate) asof_dt,
       df.tablespace_name tablespace_name,
       max(df.autoextensible) auto_ext,
       round(df.maxbytes / (1024 * 1024), 0) max_ts_size,
       round((df.bytes - sum(fs.bytes)) / (df.maxbytes) * 100, 0) max_ts_pct_used,
       round(df.bytes / (1024 * 1024), 0) curr_ts_size,
       round((df.bytes - sum(fs.bytes)) / (1024 * 1024), 0) used_ts_size,
       round((df.bytes-sum(fs.bytes)) * 100 / df.bytes, 0) ts_pct_used,
       sum(fs.bytes) / (1024 * 1024) free_ts_size,
       nvl(round(sum(fs.bytes) * 100 / df.bytes), 0) ts_pct_free
FROM dba_free_space fs,
(
    select tablespace_name,
           sum(bytes) bytes,
           sum(decode(maxbytes, 0, bytes, maxbytes)) maxbytes,
           max(autoextensible) autoextensible
    from dba_data_files
    group by tablespace_name
) df
WHERE fs.tablespace_name (+) = df.tablespace_name
GROUP BY df.tablespace_name, df.bytes, df.maxbytes

UNION ALL

SELECT trunc(sysdate) asof_dt,
       df.tablespace_name tablespace_name,
       max(df.autoextensible) auto_ext,
       round(df.maxbytes / (1024 * 1024), 0) max_ts_size,
       round((df.bytes - sum(fs.bytes)) / (df.maxbytes) * 100, 0) max_ts_pct_used,
       round(df.bytes / (1024 * 1024), 0) curr_ts_size,
       round((df.bytes - sum(fs.bytes)) / (1024 * 1024), 0) used_ts_size,
       round((df.bytes-sum(fs.bytes)) * 100 / df.bytes, 0) ts_pct_used,
       sum(fs.bytes) / (1024 * 1024) free_ts_size,
       nvl(round(sum(fs.bytes) * 100 / df.bytes), 0) ts_pct_free
FROM (
    select tablespace_name,
           bytes_free,
           bytes_used
    from v$temp_space_header
) fs,
(
    select tablespace_name,
           sum(bytes) bytes,
           sum(decode(maxbytes, 0, bytes, maxbytes)) maxbytes,
           max(autoextensible) autoextensible
    from dba_temp_files
    group by tablespace_name
) df
WHERE fs.tablespace_name (+) = df.tablespace_name
GROUP BY df.tablespace_name, df.bytes, df.maxbytes
ORDER BY 4 DESC
/

SELECT
    tf.tablespace_name,
    ROUND(SUM(tf.bytes_used) / 1024 / 1024 / 1024, 2) AS used_gb,
    ROUND(SUM(tf.bytes_free) / 1024 / 1024 / 1024, 2) AS free_gb,
    ROUND(SUM(tf.bytes_used + tf.bytes_free) / 1024 / 1024 / 1024, 2) AS current_size_gb,
    ROUND(SUM(df.maxbytes) / 1024 / 1024 / 1024, 2) AS max_size_gb,
    ROUND(SUM(tf.bytes_used) / SUM(df.maxbytes) * 100, 2) AS pct_used_of_max
FROM v$temp_space_header tf
JOIN dba_temp_files df
  ON tf.tablespace_name = df.tablespace_name
 AND tf.file_id = df.file_id
GROUP BY tf.tablespace_name
ORDER BY pct_used_of_max DESC;

            TO_CHAR(systimestamp, 'tzr')
        )
        AT TIME ZONE sessiontimezone,
        'YYYY-MM-DD HH24:MI:SS TZHTZM'
    ) AS time,

    md.cpu_time_ps AS db_cpu_users,

    SELECT DISTINCT wait_class_id
    FROM v$event_name
    WHERE wait_class = 'User I/O'
      AND 'TRUE' = 'TRUE'
) wcn,
(
    SELECT wait_class_id,
           SUM(intsize_csec) AS intsize_csec,
           MAX(end_time) AS end_time,
           SUM(time_waited) / SUM(intsize_csec) AS users
    FROM v$waitclassmetric_history
    WHERE end_time <= SYSDATE
      AND end_time > SYSDATE - 10/(60*24)
    GROUP BY wait_class_id
) wcd,
(
    SELECT
        SUM(intsize_csec) AS intsize_csec,
        MAX(end_time) AS end_time,

        AVG(
            CASE
                WHEN metric_name = 'Database Wait Time Ratio'
                THEN value
                ELSE NULL
            END
        ) AS user_wait_time_pct,

        AVG(
            CASE
                WHEN metric_name = 'Database Time Per Sec'
                THEN value / 100
                ELSE NULL
            END
        ) AS db_time_ps,

        AVG(
            CASE
                WHEN metric_name = 'CPU Usage Per Sec'
                THEN value / 100
                ELSE NULL
            END
        ) AS cpu_time_ps,

        AVG(
            CASE
                WHEN metric_name = 'Host CPU Utilization (%)'
                THEN value
                ELSE NULL
            END
        ) AS host_cpu_usage_pct

    FROM v$sysmetric_history
    WHERE metric_name IN (
        'Database Wait Time Ratio',
        'Database Time Per Sec',
        'CPU Usage Per Sec',
        'Host CPU Utilization (%)'
    )
      AND group_id = 2
      AND end_time > SYSDATE - 10/(60*24)
      AND end_time <= SYSDATE
) md
WHERE wcn.wait_class_id = wcd.wait_class_id
  AND wcd.end_time = md.end_time
  AND 'FULL' != 'BASIC'
ORDER BY md.end_time ASC;

PROMPT =========================================================
PROMPT === CDB CPU SATURATION - AUTHORITATIVE METRIC ===
PROMPT =========================================================

PROMPT PURPOSE:
PROMPT - Determine whether the CONTAINER (CDB / Exadata DB nodes) is CPU oversubscribed
PROMPT - This metric is INSTANCE-SCOPED, not PDB-scoped
PROMPT - It ALWAYS returns data and is collected by the Oracle kernel time model

PROMPT HOW TO INTERPRET (MANDATORY):
PROMPT - DB TIME represents total database demand (running + waiting)
PROMPT - DB CPU represents actual CPU service delivered
PROMPT - CPU WAIT = DB TIME - DB CPU

PROMPT JUDGMENT RULES:
PROMPT - CPU_WAIT_SEC = 0 -> GREEN (no CPU contention)
PROMPT - CPU_WAIT_SEC small and stable -> YELLOW (mild contention)
PROMPT - CPU_WAIT_SEC large or growing across instances -> RED (CDB CPU saturation)

PROMPT IMPORTANT:
PROMPT - This condition is NOT caused by any single PDB
PROMPT - When CPU_WAIT_SEC is significant on multiple instances,
PROMPT   PDB starvation is a downstream EFFECT of CDB oversubscription

PROMPT ACTION GUIDANCE:
PROMPT - Do NOT blame plans, locks, or individual PDBs when this is RED
PROMPT - Expected causes include:
PROMPT   * Concurrent ETL / PX workloads
PROMPT   * Batch overlap
PROMPT   * Insufficient CPU allocation for aggregate demand

PROMPT =========================================================

SELECT
    inst_id,

    ROUND(
        SUM(
            CASE
                WHEN stat_name = 'DB time'
                THEN value
            END
        ) / 1e6,
        2
    ) AS db_time_sec,

    ROUND(
        SUM(
            CASE
                WHEN stat_name = 'DB CPU'
                THEN value
            END
        ) / 1e6,
        2
    ) AS db_cpu_sec,

    ROUND(
        (
            SUM(
                CASE
                    WHEN stat_name = 'DB time'
                    THEN value
                END
            )
            -
            SUM(
                CASE
                    WHEN stat_name = 'DB CPU'
                    THEN value
                END
            )
        ) / 1e6,
        2
    ) AS cpu_wait_sec

FROM gv$sys_time_model
GROUP BY inst_id
ORDER BY inst_id;
